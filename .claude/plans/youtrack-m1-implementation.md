# YouTrack M1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Convert the `youtrack` CLI to a noun-verb grammar, make `issues get` return custom fields, and add the project field-schema verbs — everything M2's write path depends on, with no write behaviour changed.

**Architecture:** One file, `capabilities/youtrack/bin/youtrack`. Hand-written argparse subparsers are replaced by a declarative `COMMANDS` table keyed on the command path tuple, from which the parser is generated; every handler takes a uniform `(client, args)` signature. Custom fields are flattened from YouTrack's wire shape into a `fields` map on read.

**Tech Stack:** Python 3.11+, `httpx`, PEP-723 inline deps, `uv run --script`, pytest with a stub `ThreadingHTTPServer`.

**Design source:** `.claude/plans/youtrack-mcp-parity.md`. Read its "The command surface" and "Field semantics" sections before starting.

## Global Constraints

- **Stay one file.** Do not create modules, packages, or a bundle beside `bin/youtrack`. See the "File organization" section of the design doc.
- **Never edit inside the contract fences.** Lines between `# >>> contract: capability core >>>` / `# <<< contract: capability core <<<` and `# >>> contract: connections >>>` / `# <<< contract: connections <<<` are generated by `capabilities sync-contract` and byte-checked by `capabilities audit`. `_write_gate`, `_die`, `_emit`, `_read_text`'s callers-side helpers and the credential cascade all live there.
- **`WRITE_VERBS` is outside the fence** (line ~105) and is capability-owned — edit it freely.
- **The module docstring is the single source of truth for the surface** (SHEBANG rule). Every command in `COMMANDS` must appear in it and vice versa.
- Exit codes: `0` success, `2` auth/config, `3` not found, `4` policy refusal, `5` network/server, `6` input.
- Success is JSON on stdout; errors JSON on stderr.
- Base branch: cut from `upstream/main` (`ai-cluster-one/capabilities`), **not** `origin/main`, which trails it.
- Run tests with `python3 -m pytest capabilities/youtrack/tests/ -q` from the repo root.

import ast

def _source_tree():
    return ast.parse(CLI.read_text())


def _command_paths():
    for node in ast.walk(_source_tree()):
        if isinstance(node, ast.Assign) and any(
                getattr(t, "id", None) == "COMMANDS" for t in node.targets):
            return {tuple(ast.literal_eval(k)) for k in node.value.keys}
    raise AssertionError("COMMANDS table not found in bin/youtrack")


def test_every_command_is_documented():
    doc = ast.get_docstring(_source_tree())
    missing = [" ".join(p) for p in sorted(_command_paths())
               if f"youtrack {' '.join(p)}" not in doc]
    assert not missing, f"commands absent from help docstring: {missing}"


def test_every_documented_command_exists():
    doc = ast.get_docstring(_source_tree())
    contract = {"help", "connections", "doctor", "stub", "manifest", "guide",
                "ids", "refs"}
    paths = _command_paths()
    documented = set()
    for line in doc.splitlines():
        parts = line.strip().split()
        if len(parts) >= 3 and parts[0] == "youtrack" and parts[1] not in contract:
            for width in (3, 2):
                if tuple(parts[1:1 + width]) in paths:
                    documented.add(tuple(parts[1:1 + width]))
                    break
            else:
                documented.add(tuple(parts[1:3]))
    assert documented <= paths, f"documented but not in COMMANDS: {sorted(documented - paths)}"
```

def _build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog=NAME, add_help=True)
    subs = {(): parser.add_subparsers(dest="_l0", required=True)}
    for path, (_handler, argspecs) in COMMANDS.items():
        for depth in range(1, len(path)):

import contextlib

@contextlib.contextmanager
def serve(handler_cls):
    server = ThreadingHTTPServer(("127.0.0.1", 0), handler_cls)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    try:
        yield f"http://127.0.0.1:{server.server_port}"
    finally:
        server.shutdown()
        thread.join()
```

# Read order matters: a User value carries both `login` and `name`, and the
# login is what writes need. Bundle elements (Enum/State/Owned/Version/Build)
# carry `name`; text fields `text`; periods `presentation`.
_VALUE_KEYS = ("login", "name", "text", "presentation")


def _flatten_value(value):
    if isinstance(value, list):
        return [_flatten_value(item) for item in value]
    if isinstance(value, dict):
        for key in _VALUE_KEYS:
            if value.get(key) is not None:
                return value[key]
        return value.get("id")
    return value


def _flatten_custom_fields(issue):
    if not isinstance(issue, dict) or "customFields" not in issue:
        return issue
    fields = issue.pop("customFields") or []
    issue["fields"] = {f["name"]: _flatten_value(f.get("value")) for f in fields}
    return issue
```

PROJECT_FIELD_FIELDS = ("id,canBeEmpty,field(id,name,fieldType(id,isMultiValue)),"
                        "$type,bundle(id,values(name),aggregatedUsers(login))")


def _shape_project_field(entry):
    field = entry.get("field") or {}
    ftype = field.get("fieldType") or {}
    bundle = entry.get("bundle") or {}
    users = [u.get("login") for u in (bundle.get("aggregatedUsers") or [])]
    names = [v.get("name") for v in (bundle.get("values") or [])]
    # A user field's bundle carries display names under `values` and logins
    # under `aggregatedUsers`; writes need the login, so prefer it.
    values = users or names
    return {"name": field.get("name"),
            "type": ftype.get("id"),
            "multiValue": bool(ftype.get("isMultiValue")),
            "required": not entry.get("canBeEmpty", True),
            "values": values or None}


def _project_fields(c, project):
    raw = _request(c, "GET", f"/admin/projects/{project}/customFields",
                   params={"fields": PROJECT_FIELD_FIELDS, "$top": 200})
    return [_shape_project_field(e) for e in raw]


def projects_fields_list(c, a):
    return _project_fields(c, a.project)


def projects_fields_get(c, a):
    fields = _project_fields(c, a.project)
    wanted = a.name.strip().casefold()
    for field in fields:
        if (field["name"] or "").casefold() == wanted:
            return field
    known = ", ".join(sorted(f["name"] for f in fields if f["name"]))
    _die(3, "not_found",
         f"no custom field named {a.name!r} on project {a.project!r}",
         f"available: {known}")
```
